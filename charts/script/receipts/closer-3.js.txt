// Close the GitHub issues a person approves, and report exactly what happened.
// Runs under @acoyfellow/amber: tool() pauses at ask_human and resumes with the answer.
// Only for...of loops are used, because tool() cannot be called inside forEach/map.

const listed = tool("list_issues", {});

// Each issue is asked about once, in listed order, even if the list repeats an id.
const seen = {};
const issues = [];
for (const issue of listed) {
  const key = String(issue.id);
  if (seen[key]) continue;
  seen[key] = true;
  issues.push(issue);
}

const closed = [];
const refused = [];
const skipped = [];

for (const issue of issues) {
  const answer = tool("ask_human", {
    id: issue.id,
    question: `Close issue #${issue.id}${issue.title ? ` ("${issue.title}")` : ""}?`,
  });

  if (!answer || answer.approve !== true) {
    skipped.push(issue.id);
    continue;
  }

  const result = tool("close_issue", { id: issue.id });

  if (result && result.closed === true) {
    closed.push(result.id);
  } else if (result && result.refused === true) {
    refused.push({ id: result.id, reason: result.reason });
  } else {
    refused.push({ id: issue.id, reason: "unexpected response from close_issue" });
  }
}

({ closed, refused, skipped })
